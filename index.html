function handleFreqInput() { const freqStr = document.getElementById('freqInput').value; const freqVal = parseFloat(freqStr); if (isNaN(freqVal)) return; const modeSelect = document.getElementById('modeInput'); const stationInput = document.getElementById('stationInput'); const radioSelect = document.getElementById('radioInput'); const antennaSelect = document.getElementById('antennaInput'); const freqKey = freqVal.toFixed(3); // --- NEW: Auto-switch Radio and Antenna for Shortwave (< 30 MHz) --- if (freqVal < 30.000) { // Safely check if Malahit and GA800 exist in your hardware lists let hasMalahit = Array.from(radioSelect.options).some(o => o.value === "Malahit DSP2"); let hasGA800 = Array.from(antennaSelect.options).some(o => o.value === "GA800"); if (hasMalahit) radioSelect.value = "Malahit DSP2"; if (hasGA800) antennaSelect.value = "GA800"; } const dictMatch = appSettings.dictionary.find(d => d.f === freqKey); if (dictMatch) { stationInput.value = dictMatch.n; if (freqVal < 10.000) modeSelect.value = "LSB"; else if (freqVal < 30.000) modeSelect.value = "USB"; else if (freqVal >= 108 && freqVal <= 137) modeSelect.value = "AM"; else if (freqVal >= 137) modeSelect.value = "FM"; } else { const existingLog = cloudLogs.find(l => parseFloat(l.freq).toFixed(3) === freqKey); if (existingLog) { // Let existing logs keep their saved mode/station if (existingLog.mode) modeSelect.value = existingLog.mode; if (!stationInput.value && existingLog.station) stationInput.value = existingLog.station; return; } } // Auto-Mode standard rules (< 10 MHz = LSB, > 10 MHz = USB) if (freqVal < 10.000) modeSelect.value = "LSB"; else if (freqVal < 30.000) modeSelect.value = "USB"; else if (freqVal >= 108.000 && freqVal <= 136.975) modeSelect.value = "AM"; else if (freqVal >= 156.000 && freqVal <= 162.025) modeSelect.value = "FM"; else if (freqVal >= 446.000 && freqVal <= 446.200) modeSelect.value = "FM"; else if (freqVal >= 144.000 && freqVal <= 146.000) modeSelect.value = "FM"; else if (freqVal >= 430.000 && freqVal <= 440.000) modeSelect.value = "FM"; else if (freqVal >= 26.965 && freqVal <= 27.991) modeSelect.value = "FM"; }
